    <!-- Script de Funcionalidad -->
    <script>
        // Cargar o inicializar estado desde localStorage
        let state = JSON.parse(localStorage.getItem('finances')) || {
            incomes: [],
            categories: [],
            savings: [],
            debts: []
        };

        function saveData() {
            localStorage.setItem('finances', JSON.stringify(state));
            render();
        }

        // --- Funciones de Inserción ---
        function addCategory(e) {
            e.preventDefault();
            const name = document.getElementById('catName').value;
            const limit = parseFloat(document.getElementById('catLimit').value);
            state.categories.push({ id: Date.now(), name, limit, spent: 0 });
            document.getElementById('budgetForm').reset();
            saveData();
        }

        function addIncome(e) {
            e.preventDefault();
            const desc = document.getElementById('incDesc').value;
            const amount = parseFloat(document.getElementById('incAmount').value);
            state.incomes.push({ id: Date.now(), desc, amount });
            document.getElementById('incomeForm').reset();
            saveData();
        }

        function addSaving(e) {
            e.preventDefault();
            const title = document.getElementById('savTitle').value;
            const amount = parseFloat(document.getElementById('savAmount').value);
            state.savings.push({ id: Date.now(), title, amount });
            document.getElementById('savingsForm').reset();
            saveData();
        }

        function addDebt(e) {
            e.preventDefault();
            const title = document.getElementById('debtTitle').value;
            const amount = parseFloat(document.getElementById('debtAmount').value);
            state.debts.push({ id: Date.now(), title, amount });
            document.getElementById('debtForm').reset();
            saveData();
        }

        // --- Borrado ---
        function removeItem(type, id) {
            state[type] = state[type].filter(item => item.id !== id);
            saveData();
        }

        // --- Actualización de Gasto en Categoría ---
        function updateSpent(id, amount) {
            const cat = state.categories.find(c => c.id === id);
            if (cat) {
                cat.spent = parseFloat(amount) || 0;
                saveData();
            }
        }

        // --- Cierre de Mes ---
        function openCloseMonthModal() {
            const totalIncomes = state.incomes.reduce((acc, i) => acc + i.amount, 0);
            const totalSpent = state.categories.reduce((acc, c) => acc + c.spent, 0);
            const surplus = totalIncomes - totalSpent;

            document.getElementById('surplusAmount').innerText = `$${surplus.toFixed(2)}`;
            document.getElementById('closeMonthModal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('closeMonthModal').classList.add('hidden');
        }

        function handleSurplus(option) {
            const totalIncomes = state.incomes.reduce((acc, i) => acc + i.amount, 0);
            const totalSpent = state.categories.reduce((acc, c) => acc + c.spent, 0);
            const surplus = totalIncomes - totalSpent;

            if (surplus <= 0) {
                alert('No hay excedentes disponibles para mover este mes.');
                closeModal();
                return;
            }

            if (option === 'nextMonth') {
                state.incomes = [{ id: Date.now(), desc: 'Sobrante del Mes Anterior', amount: surplus }];
            } else if (option === 'savings') {
                if (state.savings.length === 0) {
                    state.savings.push({ id: Date.now(), title: 'Ahorro General', amount: surplus });
                } else {
                    state.savings[0].amount += surplus;
                }
                state.incomes = [];
            }

            // Reiniciar gastos del mes
            state.categories.forEach(c => c.spent = 0);
            closeModal();
            saveData();
            alert('¡El mes ha sido cerrado exitosamente!');
        }

        // --- Renderizado General ---
        function render() {
            // Totales
            const totalInc = state.incomes.reduce((acc, i) => acc + i.amount, 0);
            const totalBud = state.categories.reduce((acc, c) => acc + c.limit, 0);
            const totalSpent = state.categories.reduce((acc, c) => acc + c.spent, 0);
            const totalSav = state.savings.reduce((acc, s) => acc + s.amount, 0);
            const totalDebt = state.debts.reduce((acc, d) => acc + d.amount, 0);

            document.getElementById('totalIncome').innerText = `$${totalInc.toFixed(2)}`;
            document.getElementById('totalBudget').innerText = `$${totalBud.toFixed(2)}`;
            document.getElementById('totalSavings').innerText = `$${totalSav.toFixed(2)}`;
            document.getElementById('totalDebts').innerText = `$${totalDebt.toFixed(2)}`;
            document.getElementById('netBalance').innerText = `$${(totalInc - totalSpent).toFixed(2)}`;

            // Categorías
            const catList = document.getElementById('categoryList');
            catList.innerHTML = state.categories.map(c => `
                <div class="bg-slate-700/50 p-3 rounded-lg border border-slate-600 space-y-2">
                    <div class="flex justify-between items-center text-sm font-semibold">
                        <span>${c.name}</span>
                        <div class="flex items-center gap-2">
                            <span class="text-xs text-slate-400">$${c.spent} de $${c.limit}</span>
                            <button onclick="removeItem('categories', ${c.id})" class="text-rose-400 hover:text-rose-300 text-xs"><i class="fa-solid fa-trash"></i></button>
                        </div>
                    </div>
                    <div class="flex items-center gap-2">
                        <input type="number" value="${c.spent}" placeholder="Gastado" onchange="updateSpent(${c.id}, this.value)" class="w-full bg-slate-800 border border-slate-600 text-xs px-2 py-1 rounded text-slate-200">
                    </div>
                </div>
            `).join('');

            // Ingresos
            const incList = document.getElementById('incomeList');
            incList.innerHTML = state.incomes.map(i => `
                <tr>
                    <td class="p-2">${i.desc}</td>
                    <td class="p-2 text-emerald-400 font-semibold">$${i.amount.toFixed(2)}</td>
                    <td class="p-2 text-right">
                        <button onclick="removeItem('incomes', ${i.id})" class="text-rose-400 hover:text-rose-300 text-xs"><i class="fa-solid fa-trash"></i></button>
                    </td>
                </tr>
            `).join('');

            // Ahorros
            const savList = document.getElementById('savingsList');
            savList.innerHTML = state.savings.map(s => `
                <div class="bg-slate-700/50 p-3 rounded-lg border border-slate-600 flex justify-between items-center text-sm">
                    <div>
                        <div class="font-semibold">${s.title}</div>
                        <div class="text-xs text-amber-400 font-bold">$${s.amount.toFixed(2)}</div>
                    </div>
                    <button onclick="removeItem('savings', ${s.id})" class="text-rose-400 hover:text-rose-300 text-xs"><i class="fa-solid fa-trash"></i></button>
                </div>
            `).join('');

            // Deudas
            const debtList = document.getElementById('debtList');
            debtList.innerHTML = state.debts.map(d => `
                <div class="bg-slate-700/50 p-3 rounded-lg border border-slate-600 flex justify-between items-center text-sm">
                    <div>
                        <div class="font-semibold">${d.title}</div>
                        <div class="text-xs text-rose-400 font-bold">$${d.amount.toFixed(2)}</div>
                    </div>
                    <button onclick="removeItem('debts', ${d.id})" class="text-rose-400 hover:text-rose-300 text-xs"><i class="fa-solid fa-trash"></i></button>
                </div>
            `).join('');
        }

        // Inicializar vista
        render();
    </script>
