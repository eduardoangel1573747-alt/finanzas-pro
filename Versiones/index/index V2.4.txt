            const password = document.getElementById('authPassword').value;

        function processSavingsMovement(e) {
            e.preventDefault();
            const boxId = parseInt(document.getElementById('modalBoxId').value);
            const action = document.getElementById('modalActionType').value;
            const source = document.getElementById('modalSource').value;
            const amount = parseFloat(document.getElementById('modalAmount').value);
            const date = document.getElementById('modalDate').value;
            const reason = document.getElementById('modalReason').value;

            const box = state.savingsBoxes.find(b => b.id === boxId);
            if (box) {
                if (action === 'subtract' && amount > box.total) {
                    alert('No puedes retirar más dinero del saldo disponible.');
                    return;
                }
                box.total = action === 'add' ? box.total + amount : box.total - amount;
                box.history.unshift({ id: Date.now(), type: action, source, amount, date, reason });
                closeSavingsModal();
                saveData();
            }
        }

        function removeItem(type, id) {

        }

        function removeSavingsHistoryItem(boxId, historyId) {
            const box = state.savingsBoxes.find(b => b.id === boxId);
            if (box) {
                box.history = box.history.filter(h => h.id !== historyId);
                box.total = box.history.reduce((acc, curr) => curr.type === 'add' ? acc + curr.amount : acc - curr.amount, 0);
                saveData();
            }
        }

            const totalIncomes = state.incomes.reduce((acc, i) => acc + i.amount, 0);

        function handleSurplus(option) {
            const totalIncomes = state.incomes.reduce((acc, i) => acc + i.amount, 0);
            const totalSpent = state.categories.reduce((acc, c) => acc + c.spent, 0);
            const surplus = totalIncomes - totalSpent;

            if (surplus <= 0) {
                alert('No hay saldo sobrante disponible.');
                closeModal();
                return;
            }

            if (option === 'nextMonth') {
                state.incomes = [{ id: Date.now(), desc: 'Sobrante Mes Anterior', amount: surplus, date: today }];
            } else if (option === 'savings') {
                if (state.savingsBoxes.length === 0) {
                    state.savingsBoxes.push({
                        id: Date.now(), title: 'Ahorro Acumulado', total: surplus,
                        history: [{ id: Date.now(), type: 'add', source: 'Cierre de Mes', amount: surplus, date: today, reason: 'Excedente mensual' }]
                    });
                } else {
                    const firstBox = state.savingsBoxes[0];
                    firstBox.total += surplus;
                    firstBox.history.unshift({ id: Date.now(), type: 'add', source: 'Cierre de Mes', amount: surplus, date: today, reason: 'Excedente mensual' });
                }
                state.incomes = [];
            }

            state.categories.forEach(c => c.spent = 0);
            closeModal();
            saveData();
            alert('¡Cierre de mes realizado correctamente!');
        }
