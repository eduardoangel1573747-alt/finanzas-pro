        function handleLogin(e) {
            e.preventDefault();
            const inputVal = document.getElementById('loginUsername').value.trim().toLowerCase();
            const password = document.getElementById('loginPassword').value;
            const errorDiv = document.getElementById('loginError');

            errorDiv.classList.add('hidden');

            // Buscar por usuario o correo
            let foundKey = Object.keys(usersDB).find(u => u === inputVal || usersDB[u].email === inputVal);

            if (!foundKey || usersDB[foundKey].password !== password) {
                errorDiv.innerText = currentLang === 'es' ? 'Usuario/Correo o contraseña incorrectos.' : 'Invalid username/email or password.';
                errorDiv.classList.remove('hidden');

        }

        function handleRegister(e) {
            e.preventDefault();
            const firstName = document.getElementById('regFirstName').value.trim();
            const lastName = document.getElementById('regLastName').value.trim();
            const username = document.getElementById('regUsername').value.trim().toLowerCase();
            const age = parseInt(document.getElementById('regAge').value);
            const email = document.getElementById('regEmail').value.trim().toLowerCase();
            const securityQuestion = document.getElementById('regSecurityQuestion').value;
            const securityAnswer = document.getElementById('regSecurityAnswer').value.trim().toLowerCase();
            const password = document.getElementById('regPassword').value;
            const confirmPassword = document.getElementById('regConfirmPassword').value;
            const terms = document.getElementById('regTerms').checked;
            const errorDiv = document.getElementById('registerError');

            errorDiv.classList.add('hidden');

            if (usersDB[username]) {
                errorDiv.innerText = currentLang === 'es' ? 'El nombre de usuario ya está registrado.' : 'Username already exists.';
                errorDiv.classList.remove('hidden');
                return;
            }

            if (Object.values(usersDB).some(u => u.email === email)) {
                errorDiv.innerText = currentLang === 'es' ? 'El correo electrónico ya está en uso.' : 'Email already in use.';
                errorDiv.classList.remove('hidden');
                return;
            }

            if (!securityQuestion || !securityAnswer) {
                errorDiv.innerText = currentLang === 'es' ? 'Por favor completa la pregunta y respuesta de seguridad.' : 'Please fill in security question and answer.';
                errorDiv.classList.remove('hidden');
                return;
            }

            if (password !== confirmPassword) {
                errorDiv.innerText = currentLang === 'es' ? 'Las contraseñas no coinciden.' : 'Passwords do not match.';
                errorDiv.classList.remove('hidden');
                return;
            }

            if (!terms) {
                errorDiv.innerText = currentLang === 'es' ? 'Debes aceptar los términos y condiciones.' : 'You must accept the terms.';
                errorDiv.classList.remove('hidden');
                return;
            }

            usersDB[username] = {
                firstName, lastName, age, email, password,
                securityQuestion, securityAnswer,
                state: { transactions: [], incomes: [], categories: [], savingsBoxes: [], debts: [], receivables: [] }
            };

            localStorage.setItem('finances_users_db', JSON.stringify(usersDB));

            document.getElementById('registerForm').reset();
            toggleAuthMode();

            const successDiv = document.getElementById('authSuccess');
            successDiv.innerText = currentLang === 'es' ? '¡Cuenta creada con éxito! Por favor inicia sesión.' : 'Account created successfully! Please log in.';
            successDiv.classList.remove('hidden');
            document.getElementById('loginUsername').value = username;
        }

        function handleForgotPassword(e) {
            e.preventDefault();
            const targetUserKey = document.getElementById('recoveryTargetUser').value;
            const answerVal = document.getElementById('forgotAnswerInput').value.trim().toLowerCase();
            const newPassword = document.getElementById('forgotNewPassword').value;
            const errorDiv = document.getElementById('forgotError');

            if (userObj.securityAnswer !== answerVal) {
                errorDiv.innerText = 'La respuesta de seguridad es incorrecta.';

                return;
            }

            // Actualizar Contraseña
            userObj.password = newPassword;
            localStorage.setItem('finances_users_db', JSON.stringify(usersDB));

            cancelForgot();

            const successDiv = document.getElementById('authSuccess');
            successDiv.innerText = '¡Contraseña restablecida con éxito! Ya puedes iniciar sesión.';
            successDiv.classList.remove('hidden');

        function addTransaction(e) {
            e.preventDefault();
            const type = document.getElementById('txType').value;
            const desc = document.getElementById('txDesc').value;
            const amount = parseFloat(document.getElementById('txAmount').value);
            const date = document.getElementById('txDate').value;
            const catName = document.getElementById('txCategoryInput').value.trim() || 'General';

            const tx = { id: Date.now(), type, desc, amount, date, category: catName };
            state.transactions.unshift(tx);

            if (type === 'income') {
                state.incomes.push({ id: Date.now(), desc, amount, date });
            } else {
                let category = state.categories.find(c => c.name.toLowerCase() === catName.toLowerCase());
                if (!category) {
                    category = { id: Date.now(), name: catName, limit: amount, spent: amount };
                    state.categories.push(category);
                } else {
                    category.spent += amount;
                }
            }

            document.getElementById('txForm').reset();
            document.querySelectorAll('input[type="date"]').forEach(input => {
                if (!input.hasAttribute('title')) input.value = today;
            });
            saveData();
        }

        function addSpentManual(id, inputElement) {
            const val = parseFloat(inputElement.value);
            if (!isNaN(val) && val !== 0) {

                if (cat) {
                    cat.spent = Math.max(0, cat.spent + val);
                    inputElement.value = '';
                    saveData();
                }
            }

        function processSavingsMovement(e) {
            e.preventDefault();
            const boxId = parseInt(document.getElementById('modalBoxId').value);
            const action = document.getElementById('modalActionType').value;
            const source = document.getElementById('modalSource').value;
            const amount = parseFloat(document.getElementById('modalAmount').value);
            const date = document.getElementById('modalDate').value;
            const reason = document.getElementById('modalReason').value;

            const box = state.savingsBoxes.find(b => b.id === boxId);
            if (box) {
                if (action === 'subtract' && amount > box.total) {
                    alert('No puedes retirar más dinero del saldo disponible.');
                    return;

                box.total = action === 'add' ? box.total + amount : box.total - amount;
                box.history.unshift({ id: Date.now(), type: action, source, amount, date, reason });
                closeSavingsModal();
                saveData();
            }
        }

        function removeSavingsHistoryItem(boxId, historyId) {
            const box = state.savingsBoxes.find(b => b.id === boxId);
            if (box) {
                box.history = box.history.filter(h => h.id !== historyId);
                box.total = box.history.reduce((acc, curr) => curr.type === 'add' ? acc + curr.amount : acc - curr.amount, 0);
                saveData();
            }
        }

        function handleSurplus(option) {
            const totalIncomes = state.incomes.reduce((acc, i) => acc + i.amount, 0);
            const totalSpent = state.categories.reduce((acc, c) => acc + c.spent, 0);
            const surplus = totalIncomes - totalSpent;

            if (surplus <= 0) {
                alert('No hay saldo sobrante disponible.');
                closeModal();
                return;
            }

            if (option === 'nextMonth') {
                state.incomes = [{ id: Date.now(), desc: 'Sobrante Mes Anterior', amount: surplus, date: today }];
            } else if (option === 'savings') {
                if (state.savingsBoxes.length === 0) {
                    state.savingsBoxes.push({
                        id: Date.now(), title: 'Ahorro Acumulado', total: surplus,
                        history: [{ id: Date.now(), type: 'add', source: 'Cierre de Mes', amount: surplus, date: today, reason: 'Excedente mensual' }]
                    });
                } else {
                    const firstBox = state.savingsBoxes[0];
                    firstBox.total += surplus;
                    firstBox.history.unshift({ id: Date.now(), type: 'add', source: 'Cierre de Mes', amount: surplus, date: today, reason: 'Excedente mensual' });
                }
                state.incomes = [];
            }

            state.categories.forEach(c => c.spent = 0);
            closeModal();
            saveData();
            alert('¡Cierre de mes realizado correctamente!');
        }
