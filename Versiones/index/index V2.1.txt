    <!-- JavaScript Scripting -->
    <script>
        const today = new Date().toISOString().split('T')[0];
        document.querySelectorAll('input[type="date"]').forEach(input => {
            if (!input.hasAttribute('title')) input.value = today;
        });

        // Estado Global
        let state = JSON.parse(localStorage.getItem('finances_v3')) || {
            incomes: [],
            categories: [],
            savingsBoxes: [],
            debts: []
        };

        function saveData() {
            localStorage.setItem('finances_v3', JSON.stringify(state));
            render();
        }

        // --- Transacciones Generales ---
        function addTransaction(e) {
            e.preventDefault();
            const type = document.getElementById('txType').value;
            const desc = document.getElementById('txDesc').value;
            const amount = parseFloat(document.getElementById('txAmount').value);
            const date = document.getElementById('txDate').value;
            const catId = document.getElementById('txCategory').value;

            if (type === 'income') {
                state.incomes.push({ id: Date.now(), desc, amount, date });
            } else {
                if (catId) {
                    const category = state.categories.find(c => c.id == catId);
                    if (category) {
                        category.spent += amount;
                    }
                }
            }

            document.getElementById('txForm').reset();
            document.querySelectorAll('input[type="date"]').forEach(input => {
                if (!input.hasAttribute('title')) input.value = today;
            });
            saveData();
        }

        function toggleTxCategory() {
            const type = document.getElementById('txType').value;
            const catSelect = document.getElementById('txCategory');
            catSelect.disabled = type === 'income';
        }

        // --- Categorías (Presupuesto Mensual Sin Fecha Inicial) ---
        function addCategory(e) {
            e.preventDefault();
            const name = document.getElementById('catName').value;
            const limit = parseFloat(document.getElementById('catLimit').value);
            state.categories.push({ id: Date.now(), name, limit, spent: 0 });
            document.getElementById('budgetForm').reset();
            saveData();
        }

        // Progresivo: suma o resta según lo ingresado
        function addSpentManual(id, inputElement) {
            const val = parseFloat(inputElement.value);
            if (!isNaN(val) && val !== 0) {
                const cat = state.categories.find(c => c.id === id);
                if (cat) {
                    cat.spent = Math.max(0, cat.spent + val);
                    inputElement.value = '';
                    saveData();
                }
            }
        }

        // --- Ingresos Directos ---
        function addDirectIncome(e) {
            e.preventDefault();
            const desc = document.getElementById('incDesc').value;
            const amount = parseFloat(document.getElementById('incAmount').value);
            const date = document.getElementById('incDate').value;
            state.incomes.push({ id: Date.now(), desc, amount, date });
            document.getElementById('incomeForm').reset();
            saveData();
        }

        // --- Deudas con Pagos/Abonos ---
        function addDebt(e) {
            e.preventDefault();
            const title = document.getElementById('debtTitle').value;
            const amount = parseFloat(document.getElementById('debtAmount').value);
            const date = document.getElementById('debtDate').value || 'Sin Fecha';
            state.debts.push({ 
                id: Date.now(), 
                title, 
                amount, 
                initialAmount: amount,
                date,
                payments: []
            });
            document.getElementById('debtForm').reset();
            saveData();
        }

        function openDebtModal(debtId) {
            const debt = state.debts.find(d => d.id === debtId);
            if (!debt) return;

            document.getElementById('modalDebtId').value = debt.id;
            document.getElementById('debtModalTitle').innerText = `Abonar a: ${debt.title}`;
            document.getElementById('debtPayDate').value = today;
            document.getElementById('debtPaymentModal').classList.remove('hidden');
        }

        function closeDebtModal() {
            document.getElementById('debtPaymentModal').classList.add('hidden');
            document.getElementById('debtPaymentForm').reset();
        }

        function processDebtPayment(e) {
            e.preventDefault();
            const debtId = parseInt(document.getElementById('modalDebtId').value);
            const amount = parseFloat(document.getElementById('debtPayAmount').value);
            const date = document.getElementById('debtPayDate').value;
            const reason = document.getElementById('debtPayReason').value;

            const debt = state.debts.find(d => d.id === debtId);
            if (debt) {
                if (amount > debt.amount) {
                    alert('El abono no puede ser superior al saldo pendiente de la deuda.');
                    return;
                }

                debt.amount -= amount;
                debt.payments.unshift({
                    id: Date.now(),
                    amount,
                    date,
                    reason
                });

                closeDebtModal();
                saveData();
            }
        }

        // --- Cajas de Ahorro ---
        function createNewSavingsBox(e) {
            e.preventDefault();
            const title = document.getElementById('boxTitle').value;
            const initialAmount = parseFloat(document.getElementById('boxInitialAmount').value) || 0;
            const date = document.getElementById('boxDate').value;

            const newBox = {
                id: Date.now(),
                title: title,
                total: initialAmount,
                history: [
                    {
                        id: Date.now() + 1,
                        type: 'add',
                        amount: initialAmount,
                        date: date,
                        reason: 'Monto inicial'
                    }
                ]
            };

            state.savingsBoxes.push(newBox);
            document.getElementById('savingsBoxForm').reset();
            saveData();
        }

        function openSavingsModal(boxId) {
            const box = state.savingsBoxes.find(b => b.id === boxId);
            if (!box) return;

            document.getElementById('modalBoxId').value = box.id;
            document.getElementById('savingsModalTitle').innerText = `Ajustar: ${box.title}`;
            document.getElementById('modalDate').value = today;
            document.getElementById('savingsMovementModal').classList.remove('hidden');
        }

        function closeSavingsModal() {
            document.getElementById('savingsMovementModal').classList.add('hidden');
            document.getElementById('savingsMovementForm').reset();
        }

        function processSavingsMovement(e) {
            e.preventDefault();
            const boxId = parseInt(document.getElementById('modalBoxId').value);
            const action = document.getElementById('modalActionType').value;
            const amount = parseFloat(document.getElementById('modalAmount').value);
            const date = document.getElementById('modalDate').value;
            const reason = document.getElementById('modalReason').value;

            const box = state.savingsBoxes.find(b => b.id === boxId);
            if (box) {
                if (action === 'subtract' && amount > box.total) {
                    alert('No puedes descontar un monto mayor al saldo actual.');
                    return;
                }

                if (action === 'add') {
                    box.total += amount;
                } else {
                    box.total -= amount;
                }

                box.history.unshift({
                    id: Date.now(),
                    type: action,
                    amount: amount,
                    date: date,
                    reason: reason
                });

                closeSavingsModal();
                saveData();
            }
        }

        // --- Borrado de Elementos ---
        function removeItem(type, id) {
            state[type] = state[type].filter(item => item.id !== id);
            saveData();
        }

        // --- Cierre de Mes ---
        function openCloseMonthModal() {
            const totalIncomes = state.incomes.reduce((acc, i) => acc + i.amount, 0);
            const totalSpent = state.categories.reduce((acc, c) => acc + c.spent, 0);
            const surplus = totalIncomes - totalSpent;

            document.getElementById('surplusAmount').innerText = `$${surplus.toFixed(2)}`;
            document.getElementById('closeMonthModal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('closeMonthModal').classList.add('hidden');
        }

        function handleSurplus(option) {
            const totalIncomes = state.incomes.reduce((acc, i) => acc + i.amount, 0);
            const totalSpent = state.categories.reduce((acc, c) => acc + c.spent, 0);
            const surplus = totalIncomes - totalSpent;

            if (surplus <= 0) {
                alert('No hay excedentes sobrantes para guardar.');
                closeModal();
                return;
            }

            if (option === 'nextMonth') {
                state.incomes = [{ id: Date.now(), desc: 'Sobrante Mes Anterior', amount: surplus, date: today }];
            } else if (option === 'savings') {
                if (state.savingsBoxes.length === 0) {
                    state.savingsBoxes.push({
                        id: Date.now(),
                        title: 'Ahorro Acumulado',
                        total: surplus,
                        history: [{ id: Date.now(), type: 'add', amount: surplus, date: today, reason: 'Excedente de cierre de mes' }]
                    });
                } else {
                    const firstBox = state.savingsBoxes[0];
                    firstBox.total += surplus;
                    firstBox.history.unshift({ id: Date.now(), type: 'add', amount: surplus, date: today, reason: 'Excedente de cierre de mes' });
                }
                state.incomes = [];
            }

            state.categories.forEach(c => c.spent = 0);
            closeModal();
            saveData();
            alert('¡El mes ha finalizado con éxito!');
        }
