        function addTransaction(e) {
            e.preventDefault();
            const tx = {
                id: Date.now(),
                type: document.getElementById('txType').value,
                desc: document.getElementById('txDesc').value,
                amount: parseFloat(document.getElementById('txAmount').value),
                date: document.getElementById('txDate').value,
                category: document.getElementById('txCategoryInput').value.trim() || 'General'
            };
            state.transactions.unshift(tx);
            if (tx.type === 'income') state.incomes.push({ id: Date.now(), desc: tx.desc, amount: tx.amount, date: tx.date });
            else {
                let cat = state.categories.find(c => c.name.toLowerCase() === tx.category.toLowerCase());
                if (!cat) state.categories.push({ id: Date.now(), name: tx.category, limit: tx.amount, spent: tx.amount });
                else cat.spent += tx.amount;
            }
            document.getElementById('txForm').reset();
            saveData();
        }

        function addCategory(e) {
            e.preventDefault();
            state.categories.push({ id: Date.now(), name: document.getElementById('catName').value, limit: parseFloat(document.getElementById('catLimit').value), spent: 0 });

            saveData();

        function processReceivablePayment(e) {
            e.preventDefault();
            const rec = state.receivables.find(r => r.id == document.getElementById('modalReceivableId').value);
            const amt = parseFloat(document.getElementById('recPayAmount').value);
            if (rec && amt <= rec.amount) {
                rec.amount -= amt;
                rec.payments.unshift({ id: Date.now(), amount: amt, date: document.getElementById('recPayDate').value, reason: document.getElementById('recPayReason').value });
                closeReceivableModal();
                saveData();
            }

        function processDebtPayment(e) {
            e.preventDefault();
            const debt = state.debts.find(d => d.id == document.getElementById('modalDebtId').value);
            const amt = parseFloat(document.getElementById('debtPayAmount').value);
            if (debt && amt <= debt.amount) {
                debt.amount -= amt;
                debt.payments.unshift({ id: Date.now(), amount: amt, date: document.getElementById('debtPayDate').value, reason: document.getElementById('debtPayReason').value });
                closeDebtModal();
                saveData();

        function processSavingsMovement(e) {
            e.preventDefault();
            const box = state.savingsBoxes.find(b => b.id == document.getElementById('modalBoxId').value);
            const action = document.getElementById('modalActionType').value;
            const amt = parseFloat(document.getElementById('modalAmount').value);
            if (box) {
                if (action === 'subtract' && amt > box.total) return alert("Saldo insuficiente");
                box.total = action === 'add' ? box.total + amt : box.total - amt;
                box.history.unshift({ id: Date.now(), type: action, amount: amt, date: document.getElementById('modalDate').value, reason: document.getElementById('modalReason').value });
                closeSavingsModal();
                saveData();
            }
        }

        function openCloseMonthModal() {
            const surplus = state.incomes.reduce((a, b) => a + b.amount, 0) - state.categories.reduce((a, b) => a + b.spent, 0);
            document.getElementById('surplusAmount').innerText = currencySymbol + surplus.toFixed(2);
            document.getElementById('closeMonthModal').classList.remove('hidden');

        function handleSurplus(opt) {
            const surplus = state.incomes.reduce((a, b) => a + b.amount, 0) - state.categories.reduce((a, b) => a + b.spent, 0);
            if (opt === 'nextMonth') state.incomes = [{ id: Date.now(), desc: 'Sobrante Mes Anterior', amount: surplus, date: today }];
            else {
                if (state.savingsBoxes.length > 0) {
                    state.savingsBoxes[0].total += surplus;
                    state.savingsBoxes[0].history.unshift({ type: 'add', amount: surplus, date: today, reason: 'Excedente Mensual' });
                }
                state.incomes = [];
            }
            state.categories.forEach(c => c.spent = 0);
            closeModal();
            saveData();
        }
